cmake_minimum_required(VERSION 3.22)
project(ValiMarkdownCore LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(markdown_core cpp/Document.cpp cpp/CodeHighlighter.cpp cpp/EditorAnalysis.cpp cpp/StreamingRepair.cpp vendor/md4c/md4c.c)
target_include_directories(markdown_core PUBLIC cpp vendor/md4c)
enable_testing()
add_executable(document_tests tests/DocumentTest.cpp)
target_link_libraries(document_tests PRIVATE markdown_core)
add_test(NAME document_tests COMMAND document_tests)
add_executable(editor_analysis_tests tests/EditorAnalysisTest.cpp)
target_link_libraries(editor_analysis_tests PRIVATE markdown_core)
add_test(NAME editor_analysis_tests COMMAND editor_analysis_tests)

include(cmake/CodeHighlight.cmake)
vali_configure_highlighting(markdown_core)
add_executable(code_highlight_tests tests/CodeHighlightTest.cpp)
target_link_libraries(code_highlight_tests PRIVATE markdown_core)
add_test(NAME code_highlight_tests COMMAND code_highlight_tests)
